    고수준 클라이언트를 먼저 확인하세요

    이 페이지는 저수준(raw) 빌더 API입니다. 대부분의 경우 Getting Started의 고수준 클라이언트(client.Listener.<오퍼레이션>(ctx, ...) 형태, 직접 인자 스타일)로 충분합니다. Execute() 체인이 직접 필요한 경우에만 이 페이지를 참고하세요.

    \ListenerAPI

    All URIs are relative to https://load-balancer.kr-central-2.kakaocloud.com

    Method HTTP request Description
    CreateListener Post /api/v1/load-balancers/listeners Create listener
    DeleteListener Delete /api/v1/load-balancers/listeners/{listener_id} Delete listener
    GetListener Get /api/v1/load-balancers/listeners/{listener_id} Get listener
    ListListeners Get /api/v1/load-balancers/listeners List listeners
    UpdateListener Put /api/v1/load-balancers/listeners/{listener_id} Update listener

    CreateListener

    CreateListenerResponse CreateListener(ctx).CreateListenerRequest(createListenerRequest).Execute()

    Create listener

    Example

    package main

import (
    "context"
    "fmt"
    "os"
    openapiclient "github.com/kakaoenterprise/kc-sdk-go/v2/services/loadbalancer"
)

func main() {
    createListenerRequest := *openapiclient.NewCreateListenerRequest(*openapiclient.NewCreateListener("LoadBalancerId_example", openapiclient.Protocol("HTTP"), int32(123))) // CreateListenerRequest | 

    configuration := openapiclient.NewConfiguration()
    apiClient := openapiclient.NewAPIClient(configuration)
    resp, r, err := apiClient.ListenerAPI.CreateListener(context.Background()).CreateListenerRequest(createListenerRequest).Execute()
    if err != nil {
        fmt.Fprintf(os.Stderr, "Error when calling `ListenerAPI.CreateListener``: %v\n", err)
        fmt.Fprintf(os.Stderr, "Full HTTP response: %v\n", r)
    }
    // response from `CreateListener`: CreateListenerResponse
    fmt.Fprintf(os.Stdout, "Response from `ListenerAPI.CreateListener`: %v\n", resp)
}
